%%-- Summary
\begin{tcolorbox}[block, title=Summary]
\small\justifying
GUI agents jointly predict an \textbf{action type}
(\textit{click, scroll, type,~\ldots}) and a \textbf{target coordinate}
from a screenshot. Existing end-to-end models entangle both in one
autoregressive decode, creating the \textbf{coherent-but-wrong} failure:
commit to the wrong action type, ground precisely within that wrong frame.

\smallskip
\textbf{We study a two-stage factorization:}
\begin{enumerate}\setlength\itemsep{2pt}\small
  \item \textbf{Stage 1 —} Frozen VLM + MLP $\to$ action type
  \item \textbf{Stage 2 —} LoRA grounder conditioned on predicted type $\to$ \texttt{(x,\,y)}
\end{enumerate}
We evaluate \textbf{five conditioning variants} across two benchmarks
with paired-bootstrap significance.
\end{tcolorbox}

\begin{tcolorbox}[greenbox]
\small\textcolor{PaloAlto}{\bfseries $\checkmark$ Supported:}\;
B \& D-hook significantly beat flat baseline when action type
is spatially informative ($p<0.002$).
Two-stage pipeline beats flat E2E ($0.271$ vs $0.255$).
\end{tcolorbox}

\begin{tcolorbox}[redbox]
\small\textcolor{Cardinal}{\bfseries $\times$ Refuted:}\;
Learned prepended embedding (D-token) is \emph{not} the best
mechanism. A plain auxiliary loss (B) wins with no inference-time
conditioning.
\end{tcolorbox}

\begin{tcolorbox}[bluebox]
\small\textcolor{Bay}{\bfseries $\bullet$ Pitfall documented:}\;
\texttt{inputs\_embeds} injection in Qwen2-VL silently bypasses
M-RoPE, producing an $8\sigma$ false negative.
\end{tcolorbox}

%%-- M-RoPE
\begin{tcolorbox}[block, title=M-RoPE Implementation Pitfall]
\small\justifying
Qwen2-VL assigns 3D position IDs to visual tokens via \texttt{input\_ids}.
An earlier implementation injected action conditioning via
\texttt{inputs\_embeds}, bypassing this path entirely —
misaligning positions for \emph{all} visual tokens.

\textbf{Effect:} Variant A appeared to beat D by $\mathbf{8\sigma}$ — a pure false negative.
\textbf{Fix:} Preserve the \texttt{input\_ids} path (both D-hook and D-token do this).
\end{tcolorbox}
